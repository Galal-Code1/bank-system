CREATE TABLE EMPLOYEE (
    EMP_ID NUMBER(10) NOT NULL,
    EMP_NAME VARCHAR2(100) NOT NULL,
    JOB_TITLE VARCHAR2(100) NOT NULL,
    PASSWORD VARCHAR2(100) NOT NULL,
    CONSTRAINT PK_EMPLOYEE PRIMARY KEY (EMP_ID)
);

CREATE TABLE ACCOUNT (
    ACCOUNT_ID NUMBER(10) NOT NULL,
    USER_NAME VARCHAR2(100) NOT NULL,
    PHONE_NUMBER VARCHAR2(20) NOT NULL,
    ID_NUMBER VARCHAR2(30) NOT NULL,
    CREATED_AT DATE NOT NULL,
    BIRTH_DATE DATE NOT NULL,
    BIRTH_PLACE VARCHAR2(100) NOT NULL,
    STATUS VARCHAR2(10) NOT NULL,

    CONSTRAINT PK_ACCOUNT PRIMARY KEY (ACCOUNT_ID),
    CONSTRAINT UQ_PHONE UNIQUE (PHONE_NUMBER),
    CONSTRAINT UQ_ID_NUMBER UNIQUE (ID_NUMBER),
    CONSTRAINT CHK_STATUS CHECK (STATUS IN ('OPEN','CLOSED'))
);

CREATE TABLE ACCOUNT_BALANCE (
    ACCOUNT_ID NUMBER(10) NOT NULL,
    YER_BALANCE NUMBER(15,2) DEFAULT 0 NOT NULL,
    SAR_BALANCE NUMBER(15,2) DEFAULT 0 NOT NULL,
    USD_BALANCE NUMBER(15,2) DEFAULT 0 NOT NULL,

    CONSTRAINT PK_ACCOUNT_BALANCE PRIMARY KEY (ACCOUNT_ID),
    CONSTRAINT FK_ACCOUNT_BALANCE
        FOREIGN KEY (ACCOUNT_ID)
        REFERENCES ACCOUNT (ACCOUNT_ID),

    CONSTRAINT CHK_YER_BALANCE CHECK (YER_BALANCE >= 0),
    CONSTRAINT CHK_SAR_BALANCE CHECK (SAR_BALANCE >= 0),
    CONSTRAINT CHK_USD_BALANCE CHECK (USD_BALANCE >= 0)
);

CREATE TABLE ACCOUNT_TRANSACTIONS (
    TRANSACTION_ID NUMBER(10) NOT NULL,
    ACCOUNT_ID NUMBER(10) NOT NULL,
    AMOUNT NUMBER(15,2) NOT NULL,
    PROCESS_TYPE VARCHAR2(20) NOT NULL,
    CURRENCY VARCHAR2(3) NOT NULL,

    CONSTRAINT PK_ACCOUNT_TRANSACTIONS PRIMARY KEY (TRANSACTION_ID),

    CONSTRAINT FK_ACCOUNT_TRANSACTION
        FOREIGN KEY (ACCOUNT_ID)
        REFERENCES ACCOUNT (ACCOUNT_ID),

    CONSTRAINT CHK_AMOUNT CHECK (AMOUNT > 0),
    CONSTRAINT CHK_PROCESS_TYPE
        CHECK (PROCESS_TYPE IN ('DEPOSIT','WITHDRAW')),
    CONSTRAINT CHK_CURRENCY
        CHECK (CURRENCY IN ('YER','SAR','USD'))
);

CREATE TABLE TRANSFERS (
    TRANSFER_ID NUMBER(10) NOT NULL,
    SENDER_NAME VARCHAR2(100) NOT NULL,
    RECEIVER_NAME VARCHAR2(100) NOT NULL,
    RECEIVER_NUMBER VARCHAR2(20) NOT NULL,
    AMOUNT NUMBER(15,2) NOT NULL,
    CURRENCY VARCHAR2(3) NOT NULL,
    TRANSFER_DATE DATE NOT NULL,

    CONSTRAINT PK_TRANSFERS PRIMARY KEY (TRANSFER_ID),

    CONSTRAINT CHK_TRANSFER_AMOUNT
        CHECK (AMOUNT > 0),

    CONSTRAINT CHK_TRANSFER_CURRENCY
        CHECK (CURRENCY IN ('YER','SAR','USD'))
);

CREATE SEQUENCE ACCOUNT_SEQ
START WITH 10000
INCREMENT BY 1
NOCACHE
NOCYCLE;

CREATE SEQUENCE ACCOUNT_TRANSACTIONS_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE
NOCYCLE;

CREATE SEQUENCE TRANSFERS_SEQ
START WITH 1
INCREMENT BY 1
NOCACHE
NOCYCLE;

CREATE OR REPLACE TRIGGER TRG_ACCOUNT
BEFORE INSERT ON ACCOUNT
FOR EACH ROW
BEGIN
    IF :NEW.ACCOUNT_ID IS NULL THEN
        SELECT ACCOUNT_SEQ.NEXTVAL
        INTO :NEW.ACCOUNT_ID
        FROM DUAL;
    END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_ACCOUNT_TRANSACTIONS
BEFORE INSERT ON ACCOUNT_TRANSACTIONS
FOR EACH ROW
BEGIN
    IF :NEW.TRANSACTION_ID IS NULL THEN
        SELECT ACCOUNT_TRANSACTIONS_SEQ.NEXTVAL
        INTO :NEW.TRANSACTION_ID
        FROM DUAL;
    END IF;
END;
/

CREATE OR REPLACE TRIGGER TRG_TRANSFERS
BEFORE INSERT ON TRANSFERS
FOR EACH ROW
BEGIN
    IF :NEW.TRANSFER_ID IS NULL THEN
        SELECT TRANSFERS_SEQ.NEXTVAL
        INTO :NEW.TRANSFER_ID
        FROM DUAL;
    END IF;
END;
/

COMMIT;
